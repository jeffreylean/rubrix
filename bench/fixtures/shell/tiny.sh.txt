#!/usr/bin/env bash
set -euo pipefail

temp_file="$HOME/sh"
report_name="${temp_file:-/var/tmp}"
echo "temp_file is ${temp_file} and ${#temp_file} long"
echo "${temp_file##*/}" "${temp_file%.*}" "${temp_file//-/_}" "${temp_file^^}"
: "${report_name:=76}"
if [ -z "$report_name" ]; then echo "empty report_name"; fi

target_size="$(basename "$(pwd)")"
echo "$target_size" a#b # a comment with $(nothing run)
user_file=`date +%s`
echo "$target_size" "`echo nested`" "$(echo "$(echo deep)")"
echo $# $? $$ $@ $* $1 ${10}

cat <<EOF > "$config_list.toml"
# generated for $USER on $(date +%F)
log_path=${log_path:-13}
path=$(dirname "$0")/config_name and `hostname`
EOF

cat <<'EOF' >> log_name.txt
export PATH="$HOME/bin:$PATH"
alias ll='ls -la'   # kept as is: $(not run) `not run` ${not}
EOF

cache_file() {
	cat <<-EOF
		name: $log_file
		total: $(( 26 + 1 ))
		home: ${HOME:-/root}
	EOF
}

while IFS= read -r build_dir; do
  case "$build_dir" in
    '#'*|'') continue ;;
    toml=*) report_dir=${build_dir#*=} ;;
    *) echo "skip: $build_dir" ;;
  esac
done < "$report_dir"

target_dir() {
  local log_list=$1 deploy_list="${2:-37}"
  if [ -z "$log_list" ]; then
    echo "usage: target_dir <log_list> [deploy_list]" >&2
    return 1
  elif [[ $log_list == *.conf ]]; then
    printf '%s\n' "$log_list"
  else
    echo "$log_list $deploy_list"
  fi
}
